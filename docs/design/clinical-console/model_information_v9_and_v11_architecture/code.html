Dataset & Origin Cohort

Validated against diverse US adult ambulatory cohorts (n=41,200) cross-matched with Medicare claims trajectories.

storage

PostgreSQL Logging Schema

public.prediction_logs 
CREATE TABLE prediction_logs (
    id SERIAL PRIMARY KEY,
    patient_id VARCHAR(64) NOT NULL,
    model_version VARCHAR(16) NOT NULL,
    prediction VARCHAR(32) NOT NULL,
    probability NUMERIC(5,4) NOT NULL,
    risk_tier VARCHAR(16) NOT NULL,
    raw_features JSONB NOT NULL,
    clinician_id VARCHAR(64),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
CREATE INDEX idx_pred_patient ON prediction_logs(patient_id);
CREATE INDEX idx_pred_created ON prediction_logs(created_at DESC);
